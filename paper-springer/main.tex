\section{Problem Formulation}\label{sec:problem}
\guide{Explain the setting in your own words (hosts, VMs, tiers, daily GB budget, horizon $H$),
then each symbol of the definitions below. Source: docs/v2/THEORY.md section 1.}

\begin{definition}[Expected benefit of moving a VM]\label{def:benefit}
For VM $i$ on host $j$, with $p_j$ the probability that host $j$ fails within $H$, $c_i$ the
interruption cost, $e_i$ the probability that the migration finishes before the failure, $m_i$
the migration cost and $w_i$ its memory in GB,
\begin{equation}
b_i = p_j\, c_i\, e_i - m_i, \qquad \text{subject to} \quad \sum_{i \in S} w_i \le B .
\label{eq:benefit}
\end{equation}
\end{definition}

\begin{equation}
C(S) = C_0 - \sum_{i \in S} b_i, \qquad C_0 = \sum_i p_j\, c_i ,
\label{eq:cost}
\end{equation}
\begin{equation}
\Delta = \frac{C_{P2} - C_{\mathrm{BAM}}}{C_{P2}} .
\label{eq:delta}
\end{equation}
\guide{State that choosing $S$ is a 0/1 knapsack and that its LP relaxation is solved by
ranking on $b_i/w_i$ (Dantzig). Give the closed-form result for when ranking beats a threshold
(THEORY.md sections 2--4) and how it was checked (Figure~\ref{fig:theory}).}

\label{tab:policies}
\begin{tabular}{lp{0.72\textwidth}}\toprule
Policy & Rule \\ \midrule
P0 & Never migrate \\
P1 & Fixed probability threshold \\
P2 & Threshold tuned on development seeds to use the budget \\
P3 & Cost-sensitive rule from the first version of the project \\
P4 (BAM) & Rank by $b_i/w_i$, move while $b_i>0$ and budget remains (Algorithm~\ref{alg:bam}) \\
P5 & Narya-inspired: nothing, live-migrate or avoid allocation, chosen by Thompson sampling \\
P6 & Guard: BAM only when evidence of benefit over P2 passes; otherwise P2 \\
P7 & Oracle \\
P8 & Random selection within the budget \\ \midrule
P0+avoid & Close the riskiest hosts to new VMs; never migrate (ADR-010, ADR-011) \\
P4-net & BAM charging destination risk, $b_i = (p_j e_i - p_{\mathrm{dst}})\,c_i - m_i$ (ADR-013) \\
P0+selective & Close the riskiest $q_t$ of hosts to tier $t$ only, $q=(2,5,10)\%$ (ADR-013) \\
\bottomrule\end{tabular}
\end{table}

\begin{algorithm}[t]
\caption{Budget-aware migration (BAM, P4), one daily decision}\label{alg:bam}
\begin{algorithmic}[1]
\Require eligible VMs with $p_j, c_i, m_i, w_i$; daily budget $B$
\State compute $e_i$ from the migration queue on each host and $b_i$ by Eq.~\eqref{eq:benefit}
\State put every VM in a max-heap keyed by $b_i / w_i$; $S \gets \emptyset$; $U \gets 0$
\While{the heap is not empty}
  \State pop VM $i$; recompute $e_i$ and $b_i$ if its host's queue changed; reinsert if stale
  \If{$b_i \le 0$} \textbf{break} \EndIf
  \If{$U + w_i \le B$ and a destination has capacity}
    \State $S \gets S \cup \{i\}$; $U \gets U + w_i$; reserve the destination
  \EndIf
\EndWhile
\State \Return $S$
\end{algorithmic}
\end{algorithm}
